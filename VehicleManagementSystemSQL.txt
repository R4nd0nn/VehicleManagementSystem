-- 连接到 vehicleManageDB 数据库

-- 1. VEHICLE 车辆信息表
DROP TABLE IF EXISTS vehicle;
CREATE TABLE IF NOT EXISTS vehicle (
    id SERIAL PRIMARY KEY,
    license_plate VARCHAR(255),
    type VARCHAR(255),
    driver_id INTEGER, 
    status INTEGER, --1 未使用, 2 使用中, 3 维护中, 4 停用, 5 车辆异常
    gps_id VARCHAR(255),
    kilometres INTEGER,
    insurance_expire_date DATE
);

-- 2. DRIVER 司机信息表
DROP TABLE IF EXISTS driver;
CREATE TABLE IF NOT EXISTS driver (
    id SERIAL PRIMARY KEY,
    client_id INTEGER,
    name VARCHAR(255),
    phone_no INTEGER,
    email VARCHAR(255),
    driver_license VARCHAR(255),
    license_type VARCHAR(255),
    class_of_vehicle VARCHAR(255),
    license_issue_date DATE,
    license_expire_date DATE,
    license_issue_state VARCHAR(255),
    status INTEGER --1 空闲, 2 工作中, 3 请假中
);

-- 3. TASK 司机任务表
DROP TABLE IF EXISTS task;
CREATE TABLE IF NOT EXISTS task (
    id SERIAL PRIMARY KEY,
    order_id INTEGER,
    driver_id INTEGER,
    vehicle_id INTEGER,
    container_id INTEGER,
    task_status INTEGER,
    task_start_time TIMESTAMP,
    task_complete_time TIMESTAMP,
    pick_up_container_time TIMESTAMP,
    delivered_container_time TIMESTAMP,
    return_container_time TIMESTAMP,
    emergency_status INTEGER,
    fee_id INTEGER
);

-- 4. EXCEPTION 任务异常记录表
DROP TABLE IF EXISTS exception;
CREATE TABLE IF NOT EXISTS exception (
    id SERIAL PRIMARY KEY,
    task_id INTEGER,
    type VARCHAR(255),
    cause VARCHAR(255)
);

-- 5. CONTAINER 集装箱信息表
DROP TABLE IF EXISTS container;
CREATE TABLE IF NOT EXISTS container (
    id SERIAL PRIMARY KEY,
    container_no VARCHAR(50) NOT NULL,
    status VARCHAR(20) NOT NULL,
    pickup_time TIMESTAMP,
    return_time TIMESTAMP,
    free_days INT DEFAULT 7,
    abnormal_status VARCHAR(50),
    abnormal_desc VARCHAR(500),
    waybill_no VARCHAR(50),
    port VARCHAR(100),
    free_expired_time INTEGER,
    created_at TIMESTAMP,
    updated_at TIMESTAMP
);

-- 6. FEE 任务费用表
DROP TABLE IF EXISTS fee;
CREATE TABLE IF NOT EXISTS fee (
    id SERIAL PRIMARY KEY,
    task_id INTEGER,
    fee_type INTEGER,
    fee_note VARCHAR(255),
    create_time TIMESTAMP,
    img_id INTEGER
);

-- 7. ORDER 订单表
DROP TABLE IF EXISTS orders;
CREATE TABLE IF NOT EXISTS orders (
    id SERIAL PRIMARY KEY,
    type INTEGER,                           -- TYPE
    start_point VARCHAR(255),               -- From (注意：原表是 s，应该是 start_point)
    end_point VARCHAR(255),                 -- To/end_point
    size VARCHAR(255),                      -- size
    container_no VARCHAR(255),              -- Container NO
    pin VARCHAR(255),                       -- pin
    customer_note VARCHAR(255),             -- 客人要求
    vessel VARCHAR(255),                    -- Vessel
    shipping_line VARCHAR(255),             -- Shipping Line
    eta DATE,                               -- ETA
    first_available DATE,                   -- First avaiable
    last_free_date DATE,                    -- Last Free Date
    client_name VARCHAR(255),               -- Client Name
    customer_address VARCHAR(255),          -- customer address
    forwarder VARCHAR(255),                 -- Customer name / Forwarder
    weight VARCHAR(50),                         -- weight (提取数字)
    invoice_id VARCHAR(50),                 -- Invoice
    noted VARCHAR(255),                     -- Noted / 备注 / 空列内容
    gate_in VARCHAR(255),                    -- Gate in (2024 sheet 独有)
    gate_out VARCHAR(255),                   -- Gate out (2024 sheet 独有)
    tt VARCHAR(255),                         -- TT (2024 sheet 独有)
    status INTEGER DEFAULT 1,               -- 1:未分配, 2:进行中, 3:已完成, 4:已取消, 5:订单异常
    process_client_id INTEGER,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    create_user_id INTEGER
);

-- 8. INVOICE 订单收支表
DROP TABLE IF EXISTS invoice;
CREATE TABLE IF NOT EXISTS invoice (
    id SERIAL PRIMARY KEY,
    cost INTEGER,
    revenue INTEGER,
    status INTEGER
);

-- 9. STAFF 员工信息表
DROP TABLE IF EXISTS staff;
CREATE TABLE IF NOT EXISTS staff (
    id SERIAL PRIMARY KEY,
    role INTEGER,
    name VARCHAR(255),
    username VARCHAR(255),
    gender INTEGER,
    age INTEGER,
    birthday DATE,
    position VARCHAR(255),
    email_address VARCHAR(255),
    phone_no VARCHAR(255),
    password VARCHAR(255)
);
