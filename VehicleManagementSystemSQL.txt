-- ============================================================
-- 1. STAFF 员工信息表
-- ============================================================
DROP TABLE IF EXISTS staff;
CREATE TABLE IF NOT EXISTS staff (
    id SERIAL PRIMARY KEY,
    role INTEGER,                                      -- 0: 管理员, 1: 股东, 2: 操作人员, 3: 财务, 4: 司机
    name VARCHAR(255),                                 -- 姓名
    username VARCHAR(255),                             -- 用户名
    gender INTEGER,                                    -- 性别
    age INTEGER,                                       -- 年龄
    birthday DATE,                                     -- 生日
    position VARCHAR(255),                             -- 职位
    email_address VARCHAR(255),                        -- 邮箱
    phone_no VARCHAR(255),                             -- 电话
    password VARCHAR(255)                              -- 密码
);

-- ============================================================
-- 2. DRIVER 司机信息表
-- ============================================================
DROP TABLE IF EXISTS driver;
CREATE TABLE IF NOT EXISTS driver (
    id SERIAL PRIMARY KEY,
    client_id INTEGER,                                 -- 员工ID
    name VARCHAR(255),                                 -- 姓名
    phone_no VARCHAR(20),                              -- 电话号码
    email VARCHAR(255),                                -- 邮箱
    driver_license VARCHAR(255),                       -- 驾照号码
    license_type VARCHAR(50),                          -- 驾照类型: MR/HR/HC/MC
    license_expire_date DATE,                          -- 驾照过期日期
    status INTEGER DEFAULT 1,                          -- 司机状态: 1空闲, 2工作中, 3请假中
    msic_card_no VARCHAR(255),                         -- MSIC card number
    msic_card_expire_date DATE,                        -- MSIC card 过期日期
    dg_license_no VARCHAR(255),                        -- DG License number
    dg_license_expire_date DATE,                       -- DG License 过期日期
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 3. VEHICLE 车辆信息表
-- ============================================================
DROP TABLE IF EXISTS vehicle;
CREATE TABLE IF NOT EXISTS vehicle (
    id SERIAL PRIMARY KEY,
    license_plate VARCHAR(255),                        -- 车牌号码
    type VARCHAR(255),                                 -- 车辆类型
    gps_id VARCHAR(255),                               -- GPS ID
    kilometres INTEGER,                                -- 里程数
    rego_expire_date DATE,                             -- Rego 过期日期
    tag_number VARCHAR(255),                           -- Tag number
    fuel_card_number VARCHAR(255),                     -- Fuel card number
    status INTEGER DEFAULT 1,                          -- 车辆状态: 1空闲, 2运输中, 3维护中, 4停用, 5车辆异常
    driver_id INTEGER,                                 -- 关联司机ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 4. TASK 司机任务表
-- ============================================================
DROP TABLE IF EXISTS task;
CREATE TABLE IF NOT EXISTS task (
    id SERIAL PRIMARY KEY,
    order_id INTEGER,                                  -- 关联订单ID
    driver_id INTEGER,                                 -- 关联司机ID
    vehicle_id INTEGER,                                -- 关联车辆ID
    container_id INTEGER,                              -- 关联集装箱ID
    start_point VARCHAR(500),                          -- 派单起点
    end_point VARCHAR(500),                            -- 派单终点
    task_status INTEGER,                               -- 任务状态: 0-未派单, 1-待接单, 2-进行中, 3-已完成
    task_start_time TIMESTAMP,                         -- 任务开始时间
    task_complete_time TIMESTAMP,                      -- 任务完成时间
    emergency_status INTEGER                           -- 紧急状态: 0-普通, 1-紧急
);

-- ============================================================
-- 5. CONTAINER 集装箱信息表
-- ============================================================
DROP TABLE IF EXISTS container;
CREATE TABLE IF NOT EXISTS container (
    id SERIAL PRIMARY KEY,
    container_no VARCHAR(50) NOT NULL UNIQUE,          -- 箱号（唯一）
    status VARCHAR(20) NOT NULL DEFAULT '空柜',        -- 箱状态：满柜/空柜
    pickup_time TIMESTAMP,                             -- 提箱时间
    return_time TIMESTAMP,                             -- 还柜时间（Last free detention）
    free_days INTEGER DEFAULT 7,                       -- 免费天数
    free_expired_time INTEGER DEFAULT 7,               -- 剩余天数（前端显示用）
    abnormal_status VARCHAR(50),                       -- 异常状态
    abnormal_desc VARCHAR(500),                        -- 异常描述
    waybill_no VARCHAR(50),                            -- 关联运单号
    port VARCHAR(100),                                 -- 港口
    customer_requirement TEXT,                         -- 客人要求
    invoice_amount DECIMAL(12,2) DEFAULT 0,            -- 发票金额
    cost_status VARCHAR(20) DEFAULT '待核算',          -- 费用状态：待核算/已核算
    created_by INTEGER,                                -- 创建人ID
    updated_by INTEGER,                                -- 更新人ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,                              -- 删除时间（NULL表示未删除）
    deleted_by INTEGER                                 -- 删除人ID
);

-- ============================================================
-- 6. ORDER 订单表（完整版）
-- ============================================================
DROP TABLE IF EXISTS orders;
CREATE TABLE IF NOT EXISTS orders (
    id SERIAL PRIMARY KEY,
    type INTEGER,                                      -- 订单类型: 1-Import, 2-Export
    start_point VARCHAR(255),                          -- From/起点
    end_point VARCHAR(255),                            -- To/终点
    size VARCHAR(255),                                 -- Container Size: 20GP/40GP/45HC等
    container_no VARCHAR(255),                         -- Container NO
    pin VARCHAR(255),                                  -- EIDO Pin
    customer_note VARCHAR(255),                        -- 客人要求/Delivery Instructions
    vessel VARCHAR(255),                               -- Vessel
    shipping_line VARCHAR(255),                        -- Shipping Line
    eta DATE,                                          -- ETA
    etd DATE,                                          -- ETD (新增)
    first_available DATE,                              -- First Free Date (FFD)
    last_free_date DATE,                               -- Free Detention Date
    last_free_date_lfd DATE,                           -- Last Free Date (LFD) (新增)
    port VARCHAR(255),                                 -- Pick up Location: DPW/PAT/HPA/Others
    empty_dehire_depot VARCHAR(255),                   -- Empty Dehire Depot (新增)
    delivery_type VARCHAR(100),                        -- Delivery Type: Store Door/Container Yard/Rail Ramp/Port/Others (新增)
    door_direction VARCHAR(100),                       -- Door Direction: Pick Up/Delivery/Pick Up & Delivery (新增)
    client_name VARCHAR(255),                          -- Client Name
    customer_address VARCHAR(255),                     -- Delivery Address
    forwarder VARCHAR(255),                            -- Forwarder Name
    booking_person VARCHAR(255),                       -- Booking Person (新增)
    extra_surcharge TEXT,                              -- Extra Surcharge Details (新增)
    weight VARCHAR(50),                                -- Weight (KG)
    invoice_id VARCHAR(50),                            -- Invoice
    noted VARCHAR(255),                                -- Noted / 备注
    gate_in VARCHAR(255),                              -- Gate in
    gate_out VARCHAR(255),                             -- Gate out
    tt VARCHAR(255),                                   -- TT
    status INTEGER DEFAULT 1,                          -- 订单状态: 1未分配, 2进行中, 3待审核, 4已完成, 5已取消, 6订单异常
    process_client_id INTEGER,                         -- 处理客户ID
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    create_user_id INTEGER                             -- 创建人ID
);

-- ============================================================
-- 7. FEE 费用明细表
-- ============================================================
DROP TABLE IF EXISTS fee;
CREATE TABLE IF NOT EXISTS fee (
    id SERIAL PRIMARY KEY,
    container_id INTEGER NOT NULL,                     -- 关联货柜ID
    order_id INTEGER NOT NULL,                         -- 关联订单ID
    fee_type VARCHAR(50) NOT NULL,                     -- 费用类型：码头费用/人工费用/油费/过桥费/其他
    fee_name VARCHAR(255) NOT NULL,                    -- 费用名称
    amount DECIMAL(12,2) NOT NULL DEFAULT 0,           -- 金额
    fee_note VARCHAR(500),                             -- 备注
    img_url VARCHAR(255),                              -- 图片URL
    created_by INTEGER,                                -- 创建人ID
    updated_by INTEGER,                                -- 更新人ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(container_id, order_id, fee_name)          -- 联合唯一约束
);

-- ============================================================
-- 8. INVOICE 订单收支表

DROP TABLE IF EXISTS invoice;

-- ============================================================
-- 9. EXCEPTION 任务异常记录表
-- ============================================================
DROP TABLE IF EXISTS exception;
CREATE TABLE IF NOT EXISTS exception (
    id SERIAL PRIMARY KEY,
    task_id INTEGER,                                   -- 关联任务ID
    type VARCHAR(255),                                 -- 异常类型
    cause VARCHAR(255)                                 -- 异常原因
);

-- ============================================================
-- 10. EXCEPTION_EVENT 异常管理表
-- ============================================================
DROP TABLE IF EXISTS exception_event;
CREATE TABLE IF NOT EXISTS exception_event (
    id SERIAL PRIMARY KEY,
    exception_no VARCHAR(50) NOT NULL UNIQUE,          -- 异常编号：EXC-20260711-001
    exception_type VARCHAR(50) NOT NULL,               -- 无箱/还箱被退/故障/延误/箱损货损/客户投诉
    related_type VARCHAR(20) NOT NULL,                 -- 订单/箱号/车辆
    related_no VARCHAR(100) NOT NULL,                  -- 关联单号
    reporter VARCHAR(100) NOT NULL,                    -- 上报人
    report_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,   -- 上报时间
    description TEXT,                                  -- 异常描述
    handler VARCHAR(100),                              -- 处理人
    action TEXT,                                       -- 处理措施
    process_remark TEXT,                               -- 处理备注
    status VARCHAR(20) DEFAULT 'pending',              -- pending/processing/completed
    complete_time TIMESTAMP,                           -- 结案时间
    created_by INTEGER,
    updated_by INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    deleted_by INTEGER
);

-- ============================================================
-- 11. YARD_SLOT 场地位置表
-- ============================================================
DROP TABLE IF EXISTS yard_slot;
CREATE TABLE IF NOT EXISTS yard_slot (
    id SERIAL PRIMARY KEY,
    slot_name VARCHAR(50) NOT NULL,                    -- 位置名称
    area_type VARCHAR(20) NOT NULL,                    -- 区域类型: container/parking
    status VARCHAR(20) DEFAULT 'empty',                -- 状态: empty/occupied/reserved

    driver_id INTEGER,                                 -- 关联司机ID
    order_id VARCHAR(50),                              -- 订单号
    in_time TIMESTAMP,                                 -- 进场时间
    remark TEXT,                                       -- 备注
    created_by INTEGER,
    updated_by INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    deleted_by INTEGER,
    UNIQUE(slot_name, area_type)
);

-- ============================================================

);

-- ============================================================
-- 13. SCHEDULE_TASK 每日看板
-- ============================================================
DROP TABLE IF EXISTS schedule_task;
CREATE TABLE IF NOT EXISTS schedule_task (
    id BIGSERIAL PRIMARY KEY,
    task_id BIGINT,                                    -- 关联 task 表的 ID
    vehicle_id BIGINT NOT NULL,                        -- 所属车辆ID
    driver_id BIGINT,                                  -- 司机ID
    task_date DATE NOT NULL,                           -- 任务日期
    task_type VARCHAR(100) NOT NULL,                   -- 任务类型
    start_point VARCHAR(500),                          -- 起始点
    end_point VARCHAR(500),                            -- 终点
    task_time TIME,                                    -- 任务时间
    customer VARCHAR(200),                             -- 客户名称
    description TEXT,                                  -- 备注/描述
    status VARCHAR(20) DEFAULT 'pending',              -- pending/in_progress/completed/cancelled
    sort_order INTEGER DEFAULT 0,                      -- 排序顺序
    created_by INTEGER,                                -- 创建人ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,    -- 创建时间
    updated_by INTEGER,                                -- 更新人ID
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,    -- 更新时间
    CONSTRAINT fk_schedule_task_vehicle_id FOREIGN KEY (vehicle_id) REFERENCES vehicle(id) ON DELETE CASCADE
);

-- ============================================================
-- 17. MENU 菜单表
-- ============================================================
DROP TABLE IF EXISTS menu;
CREATE TABLE IF NOT EXISTS menu (
    menu_id BIGSERIAL PRIMARY KEY,
    menu_name VARCHAR(50) NOT NULL,                    -- 菜单名称
    parent_id BIGINT DEFAULT 0,                        -- 父菜单ID
    order_num INTEGER DEFAULT 0,                       -- 排序顺序
    path VARCHAR(200) DEFAULT '',                      -- 路由路径
    component VARCHAR(255),                            -- 组件
    query VARCHAR(255),                                -- 查询参数
    route_name VARCHAR(50) DEFAULT '',                 -- 路由名称
    menu_type CHAR(1) DEFAULT '',                      -- 菜单类型
    visible CHAR(1) DEFAULT '0',                       -- 可见性
    status CHAR(1) DEFAULT '0',                        -- 状态
    perms VARCHAR(100),                                -- 权限标识
    icon VARCHAR(100) DEFAULT '#',                     -- 图标
    remark VARCHAR(500) DEFAULT ''                     -- 备注
);
