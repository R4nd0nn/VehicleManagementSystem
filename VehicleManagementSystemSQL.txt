-- 连接到 vehicleManageDB 数据库

-- 1. VEHICLE 车辆信息表
DROP TABLE IF EXISTS vehicle;
CREATE TABLE IF NOT EXISTS vehicle (
    id SERIAL PRIMARY KEY,
    license_plate VARCHAR(255),           -- 1. 车牌号码
    type VARCHAR(255),                    -- 2. 车辆类型: semi trailer/B double/A double/retractable trailer/light trailer/Sideloader/Quad trailer/Low Loader
    gps_id VARCHAR(255),                  -- 3. GPS ID
    kilometres INTEGER,                   -- 4. 里程数
    rego_expire_date DATE,                -- 5. Rego 过期日期 (新增)
    tag_number VARCHAR(255),              -- 6. Tag number (新增)
    fuel_card_number VARCHAR(255),        -- 7. Fuel card number (新增)
    status INTEGER DEFAULT 1,             -- 8. 车辆状态: 1 未使用, 2 使用中, 3 维护中, 4 停用, 5 车辆异常
    driver_id INTEGER,                    -- 9. 关联司机ID
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. DRIVER 司机信息表
DROP TABLE IF EXISTS driver;
CREATE TABLE IF NOT EXISTS driver (
    id SERIAL PRIMARY KEY,
    client_id INTEGER,                    -- 员工ID
    name VARCHAR(255),                    -- 姓名
    phone_no VARCHAR(20),                 -- 电话号码
    email VARCHAR(255),                   -- 邮箱
    driver_license VARCHAR(255),          -- 驾照号码
    license_type VARCHAR(50),             -- 驾照类型: MR/HR/HC/MC
    license_expire_date DATE,             -- 驾照过期日期
    msic_card_no VARCHAR(255),            -- MSIC card number
    msic_card_expire_date DATE,           -- MSIC card 过期日期
    dg_license_no VARCHAR(255),           -- DG License number
    dg_license_expire_date DATE,          -- DG License 过期日期
    status INTEGER DEFAULT 1,             -- 司机状态: 1空闲, 2工作中, 3请假中
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- 3. TASK 司机任务表
DROP TABLE IF EXISTS task;
CREATE TABLE IF NOT EXISTS task (
    id SERIAL PRIMARY KEY,
    order_id INTEGER,
    driver_id INTEGER,
    vehicle_id INTEGER,
    container_id INTEGER,
    task_status INTEGER,
    task_start_time TIMESTAMP,
    task_complete_time TIMESTAMP,
    pick_up_container_time TIMESTAMP,
    delivered_container_time TIMESTAMP,
    return_container_time TIMESTAMP,
    emergency_status INTEGER,
    fee_id INTEGER
);

-- 4. EXCEPTION 任务异常记录表
DROP TABLE IF EXISTS exception;
CREATE TABLE IF NOT EXISTS exception (
    id SERIAL PRIMARY KEY,
    task_id INTEGER,
    type VARCHAR(255),
    cause VARCHAR(255)
);

-- 5. CONTAINER 集装箱信息表
DROP TABLE IF EXISTS container;
CREATE TABLE IF NOT EXISTS container (
    id SERIAL PRIMARY KEY,
    container_no VARCHAR(50) NOT NULL UNIQUE,              -- 箱号（唯一）
    status VARCHAR(20) NOT NULL DEFAULT '空柜',            -- 箱状态：满柜/空柜
    pickup_time TIMESTAMP,                                 -- 提箱时间
    return_time TIMESTAMP,                                 -- 还柜时间（Last free detention）
    free_days INT DEFAULT 7,                               -- 免费天数
    free_expired_time INT DEFAULT 7,                       -- 剩余天数（前端显示用）
    abnormal_status VARCHAR(50),                           -- 异常状态
    abnormal_desc VARCHAR(500),                            -- 异常描述
    waybill_no VARCHAR(50),                                -- 关联运单号
    port VARCHAR(100),                                     -- 港口
    customer_requirement TEXT,                             -- 客人要求
    created_by INT,                                        -- 创建人ID
    updated_by INT,                                        -- 更新人ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,                                  -- 删除时间（NULL表示未删除）
    deleted_by INT                                         -- 删除人ID
);

-- 6. FEE 任务费用表
DROP TABLE IF EXISTS fee;
CREATE TABLE IF NOT EXISTS fee (
    id SERIAL PRIMARY KEY,
    task_id INTEGER,
    fee_type INTEGER,
    fee_note VARCHAR(255),
    create_time TIMESTAMP,
    img_id INTEGER
);

-- 7. ORDER 订单表
DROP TABLE IF EXISTS orders;
CREATE TABLE IF NOT EXISTS orders (
    id SERIAL PRIMARY KEY,
    type INTEGER,                           -- TYPE
    start_point VARCHAR(255),               -- From (注意：原表是 s，应该是 start_point)
    end_point VARCHAR(255),                 -- To/end_point
    size VARCHAR(255),                      -- size
    container_no VARCHAR(255),              -- Container NO
    pin VARCHAR(255),                       -- pin
    customer_note VARCHAR(255),             -- 客人要求
    vessel VARCHAR(255),                    -- Vessel
    shipping_line VARCHAR(255),             -- Shipping Line
    eta DATE,                               -- ETA
    first_available DATE,                   -- First avaiable
    last_free_date DATE,                    -- Last Free Date
    client_name VARCHAR(255),               -- Client Name
    customer_address VARCHAR(255),          -- customer address
    forwarder VARCHAR(255),                 -- Customer name / Forwarder
    weight VARCHAR(50),                         -- weight (提取数字)
    invoice_id VARCHAR(50),                 -- Invoice
    noted VARCHAR(255),                     -- Noted / 备注 / 空列内容
    gate_in VARCHAR(255),                    -- Gate in (2024 sheet 独有)
    gate_out VARCHAR(255),                   -- Gate out (2024 sheet 独有)
    tt VARCHAR(255),                         -- TT (2024 sheet 独有)
    status INTEGER DEFAULT 1,               -- 1:未分配, 2:进行中, 3:已完成, 4:已取消, 5:订单异常
    process_client_id INTEGER,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    create_user_id INTEGER
);

-- 8. INVOICE 订单收支表
DROP TABLE IF EXISTS invoice;
CREATE TABLE IF NOT EXISTS invoice (
    id SERIAL PRIMARY KEY,
    cost INTEGER,
    revenue INTEGER,
    status INTEGER
);

-- 9. STAFF 员工信息表
DROP TABLE IF EXISTS staff;
CREATE TABLE IF NOT EXISTS staff (
    id SERIAL PRIMARY KEY,
    role INTEGER, -- 0: 管理员, 1: 股东, 2: 操作人员, 3: 财务, 4: 司机
    name VARCHAR(255),
    username VARCHAR(255),
    gender INTEGER,
    age INTEGER,
    birthday DATE,
    position VARCHAR(255),
    email_address VARCHAR(255),
    phone_no VARCHAR(255),
    password VARCHAR(255)
);

-- 10. YARD_SLOT 场地位置表
DROP TABLE IF EXISTS yard_slot;
CREATE TABLE IF NOT EXISTS yard_slot (
    id SERIAL PRIMARY KEY,
    slot_name VARCHAR(50) NOT NULL,                    -- 位置名称: C1, C2, Trailer 1, Trailer 2...
    area_type VARCHAR(20) NOT NULL,                    -- 区域类型: container / parking
    status VARCHAR(20) DEFAULT 'empty',                -- 状态: empty / occupied / reserved
    container_no VARCHAR(50),                          -- 货柜号（占用时有值）
    order_id VARCHAR(50),                              -- 关联订单号
    driver_name VARCHAR(100),                          -- 关联司机姓名
    in_time TIMESTAMP,                                 -- 进场时间
    remark TEXT,                                       -- 备注
    created_by INT,                                    -- 创建人ID
    updated_by INT,                                    -- 更新人ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,                              -- 软删除
    deleted_by INT,                                    -- 删除人ID
    UNIQUE(slot_name, area_type)                       -- 同一区域内名称唯一
);

-- 11. YARD_LOG 场地操作日志表
DROP TABLE IF EXISTS yard_log;
CREATE TABLE IF NOT EXISTS yard_log (
    id SERIAL PRIMARY KEY,
    slot_id INT NOT NULL,                              -- 关联位置ID
    slot_name VARCHAR(50) NOT NULL,                    -- 位置名称（冗余，方便查询）
    area_type VARCHAR(20) NOT NULL,                    -- 区域类型（冗余）
    action_type VARCHAR(30) NOT NULL,                  -- in / out / reserve / cancel_reserve / add / delete / edit
    container_no VARCHAR(50),                          -- 货柜号
    order_id VARCHAR(50),                              -- 订单号
    driver_name VARCHAR(100),                          -- 司机姓名
    remark TEXT,                                       -- 备注
    operator_id INT,                                   -- 操作人ID
    operator_name VARCHAR(100),                        -- 操作人姓名
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 12. 每日看板
DROP TABLE IF EXISTS schedule_task;
CREATE TABLE IF NOT EXISTS schedule_task (
    id BIGSERIAL PRIMARY KEY,
    vehicle_id BIGINT NOT NULL,                          -- 所属车辆ID
    task_date DATE NOT NULL,                             -- 任务日期
    task_type VARCHAR(100) NOT NULL,                     -- 任务类型（用户自定义输入，如：提箱、送箱、还箱）
    start_point VARCHAR(500),                            -- 起始点（原 location 字段改名）
    end_point VARCHAR(500),                              -- 终点（新增）
    task_time TIME,                                      -- 任务时间
    customer VARCHAR(200),                               -- 客户名称
    description TEXT,                                    -- 备注/描述
    status VARCHAR(20) DEFAULT 'pending',                -- 状态：pending/in_progress/completed/cancelled
    sort_order INT DEFAULT 0,                            -- 排序顺序
    created_by INT,                                      -- 创建人ID
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,      -- 创建时间
    updated_by INT,                                      -- 更新人ID
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP       -- 更新时间
);

-- 13. 月度收支数据
DROP TABLE IF EXISTS monthly_expense_detail;
CREATE TABLE IF NOT EXISTS monthly_expense_detail (
    id SERIAL PRIMARY KEY,
    expense_month VARCHAR(7) NOT NULL,      -- 月份
    label VARCHAR(100) NOT NULL,            -- 费用项目名称
    amount DECIMAL(12,2) NOT NULL,          -- 金额
    remark VARCHAR(500),                    -- 备注
    is_fixed BOOLEAN DEFAULT FALSE,         -- 是否固定项
    created_by INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
